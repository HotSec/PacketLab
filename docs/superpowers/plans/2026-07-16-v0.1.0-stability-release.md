```go
	}
	skipSuffixes := []string{

		".settings.data.microsoft.com",
```

```go
func readBody(req *http.Request) string {
	if req.Body == nil {
		return ""
	}
	// 优先 GetBody（OnRequest 已设置，可重复读）
	if req.GetBody != nil {
		if body, err := req.GetBody(); err == nil {
			defer body.Close()
			raw, err := io.ReadAll(io.LimitReader(body, 64*1024))
			if err == nil && len(raw) > 0 {
				return string(raw)
			}
		}
	}
	// fallback：直接读 req.Body（会消费，仅在未走 GetBody 路径时）
	raw, err := io.ReadAll(io.LimitReader(req.Body, 64*1024))
	if err != nil || len(raw) == 0 {
		return ""
	}
	return string(raw)
}
```

注意：Task 1 的修复已让 `proxy.go OnRequest` 设置 `req.GetBody`，此处直接消费即可。fallback 分支保留兼容性（如直接构造 req 调用 interceptor 的测试场景）。

```go
package capture

import (
	"sync"
	"testing"
	"time"
)

// TestMemRingBuffer_Concurrent 8 producer + 1 consumer race 检测
func TestMemRingBuffer_Concurrent(t *testing.T) {
	ring := NewMemRingBuffer(1024)
	var wg sync.WaitGroup

	// 8 producer
	for i := 0; i < 8; i++ {
		wg.Add(1)
		go func(p int) {
			defer wg.Done()
			for j := 0; j < 1000; j++ {
				ring.Push(Record{ID: int64(p*1000 + j)})
			}
		}(i)
	}

	// 1 consumer
	done := make(chan struct{})
	go func() {
		for {
			batch := ring.PopBatch(100)
			if len(batch) == 0 {
				time.Sleep(time.Millisecond)
			}
			select {
			case <-done:
				return
			default:
			}
		}
	}()

	wg.Wait()
	ring.Stop()
	close(done)

	if ring.Overwrites() > 0 {
		t.Logf("overwrites: %d (expected 0 or small)", ring.Overwrites())
	}
}

func TestMemRingBuffer_PushAfterClose(t *testing.T) {
	ring := NewMemRingBuffer(64)
	ring.Stop()
	if ring.Push(Record{ID: 1}) {
		t.Fatal("expected Push to return false after Stop")
	}
}

func TestMemRingBuffer_Overwrite(t *testing.T) {
	ring := NewMemRingBuffer(8) // 容量 8
	for i := 0; i < 16; i++ {
		ring.Push(Record{ID: int64(i)})
	}
	if ring.Overwrites() == 0 {
		t.Fatal("expected overwrites > 0")
	}
}

func TestMemRingBuffer_PopBatchBlocked(t *testing.T) {
	ring := NewMemRingBuffer(64)
	started := make(chan struct{})
	done := make(chan []Record, 1)

	go func() {
		close(started)
		batch := ring.PopBatch(10)
		done <- batch
	}()

	<-started
	time.Sleep(50 * time.Millisecond)
	ring.Push(Record{ID: 42})

	select {
	case batch := <-done:
		if len(batch) != 1 || batch[0].ID != 42 {
			t.Fatalf("unexpected batch: %v", batch)
```

完整替换 `internal/capture/memring.go` 的 MemRingBuffer 实现：

```go
package capture

import (
	"sync"
	"sync/atomic"
)

type MemRingBuffer struct {
	mu          sync.Mutex
	cv          *sync.Cond
	buf         []Record
	head, tail  uint64
	mask        uint64
	overwrites  atomic.Uint64
	stopped     bool
}

func NewMemRingBuffer(size int) *MemRingBuffer {
	if size <= 0 {
		size = 262144
	}
	// 向上取 2 的幂
	size = nextPow2(size)
	r := &MemRingBuffer{
		buf:  make([]Record, size),
		mask: uint64(size - 1),
	}
	r.cv = sync.NewCond(&r.mu)
	return r
}

func nextPow2(n int) int {
	x := 1
	for x < n {
		x <<= 1
	}
	return x
}

func (r *MemRingBuffer) Push(rec Record) bool {
	r.mu.Lock()
	defer r.mu.Unlock()
	if r.stopped {
		return false
	}
	next := (r.head + 1) & r.mask
	if next == r.tail {
		// 满，淘汰最老
		r.tail = (r.tail + 1) & r.mask
		r.overwrites.Add(1)
	}
	r.buf[r.head] = rec
	r.head = next
	r.cv.Signal()
	return true
}

func (r *MemRingBuffer) PopBatch(max int) []Record {
	r.mu.Lock()
	defer r.mu.Unlock()

	for r.head == r.tail && !r.stopped {
		r.cv.Wait()
	}

	available := (r.head - r.tail) & r.mask
	if available == 0 {
		return nil // stopped
	}
	if uint64(max) > available {
		max = int(available)
	}

	out := make([]Record, max)
	for i := 0; i < max; i++ {
		out[i] = r.buf[r.tail]
		r.tail = (r.tail + 1) & r.mask
	}
	return out
}

func (r *MemRingBuffer) Stop() {
	r.mu.Lock()
	defer r.mu.Unlock()
	r.stopped = true
	r.cv.Broadcast()
}

func (r *MemRingBuffer) Overwrites() uint64 {
	return r.overwrites.Load()
}

func (r *MemRingBuffer) Size() int {
	r.mu.Lock()
	defer r.mu.Unlock()
	return int((r.head - r.tail) & r.mask)
}
```

注意：`Record` 类型保持现有定义（在 memring.go 或 engine.go 中）。

```go
	query := `
		SELECT host, COUNT(*) as cnt FROM (
			SELECT DISTINCT host FROM requests
		) WHERE 1=1`

	if search != "" {
		query += " AND host LIKE ?"
		args = append(args, "%"+search+"%")
	}
	query += " GROUP BY host ORDER BY cnt DESC LIMIT ? OFFSET ?"
	args = append(args, limit, offset)

	rows, err := s.readDB().Query(query, args...)
	// ... 解析 rows ...
```

```go
func (s *streamPool) evictOldestIdleStream() {
	s.assembler.mu.Lock()
	defer s.assembler.mu.Unlock()

	var oldestKey flowKey
	var oldestTime time.Time
	for k, stream := range s.assembler.streams {
```

```go
func TestWS_BroadcastMultipleClients(t *testing.T) {
	hub := NewHub()
	go hub.Run()
	defer hub.Stop()

	srv := httptest.NewServer(http.HandlerFunc(hub.handleWS))
	defer srv.Close()

	url := "ws" + strings.TrimPrefix(srv.URL, "http")
	c1, _, _ := websocket.DefaultDialer.Dial(url, nil)
	c2, _, _ := websocket.DefaultDialer.Dial(url, nil)
	defer c1.Close()
	defer c2.Close()

	// 等待注册
	time.Sleep(50 * time.Millisecond)

	hub.Broadcast([]byte(`{"type":"new_request"}`))

	for _, c := range []*websocket.Conn{c1, c2} {
		c.SetReadDeadline(time.Now().Add(time.Second))
		_, msg, err := c.ReadMessage()
		if err != nil {
			t.Fatal(err)
		}
		if !strings.Contains(string(msg), "new_request") {
			t.Fatalf("client did not receive: %s", msg)
		}
	}
}

	// 不读，让缓冲快速填满
	for i := 0; i < 100; i++ {
		// 应该不阻塞
		hub.Broadcast([]byte(`{"type":"new_request"}`))
```
